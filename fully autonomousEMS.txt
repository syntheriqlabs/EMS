import uuid, time, statistics, re
from dataclasses import dataclass, field
from typing import List, Dict

# --- INTERFACES ---
@dataclass
class EMSRequest:
    content: str
    tags: Dict[str, List[str]]
    entities: List[str]

@dataclass
class EMSResponse:
    decision: str
    content: str
    tension_index: float
    audit_log: str
    policy_note: str

@dataclass
class EthicalMemory:
    content: str
    id: str = field(default_factory=lambda: str(uuid.uuid4()))
    timestamp: float = field(default_factory=time.time)
    cvl_score: float; cel_score: float; ril_score: float; overall_ethical_score: float
    tension_index: float; is_ethically_valid: bool; tension_type: str
    cvl_tags: List[str]; cel_context: List[str]

# --- SENSORY LAYERS ---
class CoreValuesLayer:
    def __init__(self): self.w = {"NON_MALEFICENCE": 10, "AUTONOMY": 9, "VERACITY": 8, "TRANSPARENCY": 7}
    def evaluate(self, tags):
        s = [self.w.get(t, 5)/10.0 for t in tags] if tags else [0.5]
        return round(statistics.mean(s), 2), s

class ContextualEthicsLayer:
    def __init__(self): self.m = {"RESEARCH": 1.2, "URGENT": 1.3, "MEDICAL": 1.5, "HIGH_RISK": 0.7}
    def evaluate(self, tags):
        mult = max([self.m.get(t, 1.0) for t in tags]) if tags else 1.0
        return round(mult/2.0, 2)

class RelationalIdentityLayer:
    def __init__(self): self.ent = {}
    def register(self, eid, t, d): self.ent[eid] = {"t": t, "d": d}
    def evaluate(self, eids):
        data = [self.ent.get(i, {"t": 0.1, "d": 0.1}) for i in eids]
        scores = [(d["t"]*0.7 + d["d"]*0.3) for d in data]
        return round(statistics.mean(scores), 2) if scores else 0.1

# --- CLASSIFIER ---
class TagClassifier:
    def __init__(self):
        self.cvl_rules = {"NON_MALEFICENCE": [r"(?i)bypass", r"(?i)exploit"], "AUTONOMY": [r"(?i)freedom", r"(?i)my rights"]}
        self.cel_rules = {"RESEARCH": [r"(?i)research"], "URGENT": [r"(?i)urgent"], "MEDICAL": [r"(?i)medical"]}
        self.ril_rules = {"ADMIN": [r"(?i)admin"], "RESEARCHER": [r"(?i)researcher"], "CHILD": [r"(?i)child"]}
        self.identity_map = {
            "ADMIN": {"t": 0.95, "d": 0.9}, "RESEARCHER": {"t": 0.75, "d": 0.5},
            "CHILD": {"t": 0.60, "d": 0.2}, "ANONYMOUS": {"t": 0.30, "d": 0.1}
        }
    def classify(self, text):
        tags = {"cvl": [], "cel": [], "ril": "ANONYMOUS"}
        for k, p in self.cvl_rules.items():
            if any(re.search(r, text) for r in p): tags["cvl"].append(k)
        for k, p in self.cel_rules.items():
            if any(re.search(r, text) for r in p): tags["cel"].append(k)
        for k, p in self.ril_rules.items():
            if any(re.search(r, text) for r in p): tags["ril"] = k; break
        if not tags["cvl"]: tags["cvl"] = ["TRANSPARENCY"]
        return tags

# --- AGENT & ENGINE ---
class EMSEngine:
    def __init__(self, cvl, cel, ril):
        self.cvl, self.cel, self.ril = cvl, cel, ril
        self.block_threshold, self.allow_threshold = 0.45, 0.75
        self.cvl_strictness, self.ril_strictness = 0.5, 0.3
    def process(self, content, tags, entities, offset):
        c_m, c_s = self.cvl.evaluate(tags["cvl"])
        e_s = self.cel.evaluate(tags["cel"])
        r_s = self.ril.evaluate(entities)
        overall = round((c_m*0.4 + e_s*0.3 + r_s*0.3), 2)
        valid = (c_m >= (self.cvl_strictness + offset)) and (r_s >= self.ril_strictness)
        return EthicalMemory(content, c_m, e_s, r_s, overall, 0.0, valid, "NONE", tags["cvl"], tags["cel"])

class EMSAgent:
    def __init__(self, engine, voice, dialogue, learner, classifier):
        self.engine, self.voice, self.dialogue, self.learner, self.classifier = engine, voice, dialogue, learner, classifier
        self.bank = []
    def handle(self, content):
        tags = self.classifier.classify(content)
        id_type = tags["ril"]
        prof = self.classifier.identity_map[id_type]
        self.engine.ril.register(id_type, prof["t"], prof["d"])
        mem = self.engine.process(content, tags, [id_type], self.dialogue.get_dynamic_strictness_offset())
        self.bank.append(mem); self.dialogue.record_turn(mem); self.learner.adapt(self.bank)
        if not mem.is_ethically_valid or mem.overall_ethical_score < self.engine.block_threshold:
            decision, final = "BLOCK", "REFUSED."
        elif mem.overall_ethical_score < self.engine.allow_threshold:
            decision, final = "MODIFY", "MODIFIED: " + mem.content
        else:
            decision, final = "ALLOW", mem.content
        return EMSResponse(decision, final, 0.0, f"ID: {id_type}", "Stable")
